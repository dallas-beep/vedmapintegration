async function loadEvents() { byId('mapStatus').textContent = 'Loading events...'; try { const response = await fetch('/api/get-mobilize-events', { headers: { Accept: 'application/json' } }); if (!response.ok) throw new Error(`API error: ${response.status}`); const payload = await response.json(); const events = Array.isArray(payload) ? payload : payload.data; if (!Array.isArray(events)) throw new Error(payload && payload.error ? payload.error : 'The events API returned no data array'); resolvedEvents = (Array.isArray(events) ? events : []).map(event => ({ id: text(event.id), title: text(event.title || 'Untitled event'), hostOrganization: text(event.hostOrganization), date: text(event.date), time: text(event.time), address: text(event.address), city: text(event.city), state: text(event.state), zip: text(event.zip), social: text(event.social), octoberEvent: text(event.octoberEvent), public: text(event.public), mobilize: text(event.mobilize), description: text(event.description), browserUrl: text(event.browserUrl), lat: Number(event.lat), lon: Number(event.lon) })).filter(event => Number.isFinite(event.lat) && Number.isFinite(event.lon)); if (payload.diagnostics) { console.info('Events import diagnostics:', payload.diagnostics); } byId('mapStatus').textContent = `${resolvedEvents.length} events live`; filterEvents(false); } catch (error) { console.error('Unable to load events:', error); byId('mapStatus').textContent = 'Error loading events'; byId('eventSidebarList').innerHTML = '
Events are temporarily unavailable.
'; } }
